    <script>
        let songs = [];
        let currentSongIndex = 0;
        let selectedScore = 0;
        let votes = {}; // 투표자별 투표 기록
        let songVotes = {}; // 곡별 투표 결과

        function addSong() {
            const title = document.getElementById('songTitle').value.trim();
            const artist = document.getElementById('artistName').value.trim();
            const performer = document.getElementById('performerName').value.trim();

            if (!title || !artist || !performer) {
                showMessage('모든 정보를 입력해주세요!', 'error');
                return;
            }

            const song = {
                id: Date.now(),
                title: title,
                artist: artist,
                performer: performer
            };

            songs.push(song);
            songVotes[song.id] = [];
            
            updateSongList();
            clearInputs();
            
            document.getElementById('startBtn').disabled = false;
            showMessage('곡이 추가되었습니다!', 'success');
        }

        function updateSongList() {
            const songList = document.getElementById('songList');
            if (songs.length === 0) {
                songList.innerHTML = '<p style="color: #666; text-align: center; padding: 20px;">등록된 곡이 없습니다.</p>';
                return;
            }

            songList.innerHTML = songs.map((song, index) => `
                <div class="song-result" style="margin-bottom: 10px;">
                    <h3>${index + 1}. ${song.title}</h3>
                    <p>가수: ${song.artist} | 참가자: ${song.performer}</p>
                    <button class="btn btn-danger" onclick="removeSong(${song.id})" style="margin-top: 10px;">삭제</button>
                </div>
            `).join('');
        }

        function removeSong(songId) {
            songs = songs.filter(song => song.id !== songId);
            delete songVotes[songId];
            updateSongList();
            
            if (songs.length === 0) {
                document.getElementById('startBtn').disabled = true;
            }
            
            showMessage('곡이 삭제되었습니다.', 'success');
        }

        function clearInputs() {
            document.getElementById('songTitle').value = '';
            document.getElementById('artistName').value = '';
            document.getElementById('performerName').value = '';
        }

        function startVoting() {
            if (songs.length === 0) {
                showMessage('먼저 곡을 등록해주세요!', 'error');
                return;
            }

            currentSongIndex = 0;
            document.getElementById('adminSection').style.display = 'none';
            document.getElementById('votingSection').style.display = 'block';
            document.getElementById('resultsSection').style.display = 'none';
            
            showCurrentSong();
        }

        function showCurrentSong() {
            if (currentSongIndex >= songs.length) {
                showResults();
                return;
            }

            const song = songs[currentSongIndex];
            document.getElementById('currentTitle').textContent = song.title;
            document.getElementById('currentArtist').textContent = `가수: ${song.artist}`;
            document.getElementById('currentPerformer').textContent = `참가자: ${song.performer}`;
            
            // 투표 초기화
            selectedScore = 0;
            document.getElementById('voterName').value = '';
            clearScoreSelection();
            document.getElementById('submitVoteBtn').disabled = true;
            
            // 다음 곡 버튼 표시 여부
            document.getElementById('nextSongBtn').style.display = 
                currentSongIndex < songs.length - 1 ? 'inline-block' : 'none';
        }

        function selectScore(score) {
            selectedScore = score;
            
            // 모든 버튼 선택 해제
            document.querySelectorAll('.score-btn').forEach(btn => {
                btn.classList.remove('selected');
            });
            
            // 선택된 버튼 활성화
            event.target.classList.add('selected');
            
            // 투표자 이름이 있으면 투표 버튼 활성화
            checkVoteReady();
        }

        function checkVoteReady() {
            const voterName = document.getElementById('voterName').value.trim();
            const submitBtn = document.getElementById('submitVoteBtn');
            
            submitBtn.disabled = !(selectedScore > 0 && voterName);
        }

        // 투표자 이름 입력 시 실시간 체크
        document.getElementById('voterName').addEventListener('input', checkVoteReady);

        function submitVote() {
            const voterName = document.getElementById('voterName').value.trim();
            const songId = songs[currentSongIndex].id;

            if (!voterName || selectedScore === 0) {
                showMessage('투표자 이름과 점수를 선택해주세요!', 'error');
                return;
            }

            // 중복 투표 방지
            const voteKey = `${voterName}_${songId}`;
            if (votes[voteKey]) {
                showMessage('이미 이 곡에 투표하셨습니다!', 'error');
                return;
            }

            // 투표 기록
            votes[voteKey] = true;
            songVotes[songId].push({
                voter: voterName,
                score: selectedScore,
                timestamp: new Date().toLocaleTimeString()
            });

            showMessage(`${voterName}님의 투표가 완료되었습니다! (${selectedScore}점)`, 'success');
            
            // 폼 초기화
            document.getElementById('voterName').value = '';
            selectedScore = 0;
            clearScoreSelection();
            document.getElementById('submitVoteBtn').disabled = true;
        }

        function clearScoreSelection() {
            document.querySelectorAll('.score-btn').forEach(btn => {
                btn.classList.remove('selected');
            });
        }

        function nextSong() {
            currentSongIndex++;
            showCurrentSong();
        }

        function showResults() {
            document.getElementById('adminSection').style.display = 'none';
            document.getElementById('votingSection').style.display = 'none';
            document.getElementById('resultsSection').style.display = 'block';
            
            calculateAndDisplayResults();
        }

        function calculateAndDisplayResults() {
            const results = songs.map(song => {
                const votes = songVotes[song.id] || [];
                const totalVotes = votes.length;
                const totalScore = votes.reduce((sum, vote) => sum + vote.score, 0);
                const averageScore = totalVotes > 0 ? (totalScore / totalVotes).toFixed(2) : 0;
                
                return {
                    ...song,
                    totalVotes,
                    totalScore,
                    averageScore: parseFloat(averageScore),
                    votes
                };
            });

            // 평균 점수로 정렬 (높은 순)
            results.sort((a, b) => b.averageScore - a.averageScore);

            const resultsGrid = document.getElementById('resultsGrid');
            resultsGrid.innerHTML = results.map((result, index) => `
                <div class="song-result ${index === 0 ? 'winner' : ''}">
                    <h3>${index === 0 ? '🏆 ' : `${index + 1}위 `}${result.title}</h3>
                    <p>가수: ${result.artist} | 참가자: ${result.performer}</p>
                    
                    <div class="result-stats">
                        <div class="stat-box">
                            <div class="stat-value">${result.averageScore}</div>
                            <div class="stat-label">평균 점수</div>
                        </div>
                        <div class="stat-box">
                            <div class="stat-value">${result.totalVotes}</div>
                            <div class="stat-label">총 투표수</div>
                        </div>
                        <div class="stat-box">
                            <div class="stat-value">${result.totalScore}</div>
                            <div class="stat-label">총점</div>
                        </div>
                    </div>
                    
                    ${result.votes.length > 0 ? `
                        <details style="margin-top: 15px;">
                            <summary style="cursor: pointer; font-weight: bold;">투표 상세 보기</summary>
                            <div style="margin-top: 10px; font-size: 0.9em;">
                                ${result.votes.map(vote => 
                                    `<p>${vote.voter}: ${vote.score}점 (${vote.timestamp})</p>`
                                ).join('')}
                            </div>
                        </details>
                    ` : ''}
                </div>
            `).join('');
        }

        function backToAdmin() {
            document.getElementById('adminSection').style.display = 'block';
            document.getElementById('votingSection').style.display = 'none';
            document.getElementById('resultsSection').style.display = 'none';
        }

        function exportResults() {
            const results = songs.map(song => {
                const votes = songVotes[song.id] || [];
                const totalVotes = votes.length;
                const totalScore = votes.reduce((sum, vote) => sum + vote.score, 0);
                const averageScore = totalVotes > 0 ? (totalScore / totalVotes).toFixed(2) : 0;
                
                return {
                    곡명: song.title,
                    가수: song.artist,
                    참가자: song.performer,
                    총투표수: totalVotes,
                    총점: totalScore,
                    평균점수: averageScore
                };
            });

            results.sort((a, b) => b.평균점수 - a.평균점수);

            let csvContent = "순위,곡명,가수,참가자,총투표수,총점,평균점수\n";
            results.forEach((result, index) => {
                csvContent += `${index + 1},"${result.곡명}","${result.가수}","${result.참가자}",${result.총투표수},${result.총점},${result.평균점수}\n`;
            });

            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = `노래경연_결과_${new Date().toLocaleDateString()}.csv`;
            link.click();
        }

        function showMessage(message, type) {
            const messageDiv = document.getElementById('voteMessage');
            messageDiv.className = type === 'error' ? 'error-message' : 'success-message';
            messageDiv.textContent = message;
            messageDiv.style.display = 'block';
            
            setTimeout(() => {
                messageDiv.style.display = 'none';
            }, 3000);
        }
    </script>
